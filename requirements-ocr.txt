# =============================================================================
# URE - Food Label Analyser :: OPTIONAL OCR dependencies
# Install only if you want to run the image -> text pipeline:
#   pip install -r requirements-ocr.txt
#
# Engine: RapidOCR (rapidocr-onnxruntime)
# -----------------------------------------------------------------------------
# PaddleOCR was the original choice, but PaddlePaddle publishes no wheels for
# Python 3.14 (this project's runtime), so it cannot be installed here.
# RapidOCR is used instead: it has cp314 wheels, needs no external system
# binary, bundles the PP-OCR models in the wheel, and produces comparable
# results (same model family, exported to ONNX).
#
# The backend runs fine WITHOUT this file installed - image OCR then reports
# "unavailable" (HTTP 501) and callers can use the manual-entry endpoint.
# =============================================================================

rapidocr-onnxruntime>=1.2.3

# --- Image prep --------------------------------------------------------------
# Used to apply EXIF rotation and cap phone-photo resolution before OCR.
# (Also a transitive dependency of RapidOCR, but listed because we import it.)
